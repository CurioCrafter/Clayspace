cmake_minimum_required(VERSION 3.22)
project(monado_projection_smoke LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows simulator-only smoke")
endif()
set(XR_SDK "${CMAKE_CURRENT_LIST_DIR}/../../dependencies/windows_x64_5_1/xr_openxr_sdk")
add_executable(monado_projection_smoke projection_smoke.cpp)
target_compile_features(monado_projection_smoke PRIVATE cxx_std_17)
target_include_directories(monado_projection_smoke PRIVATE "${XR_SDK}/include")
target_link_libraries(monado_projection_smoke PRIVATE "${XR_SDK}/lib/openxr_loader.lib" opengl32 gdi32 user32)

add_executable(index_runtime_system_probe index_runtime_system_probe.cpp)
target_compile_features(index_runtime_system_probe PRIVATE cxx_std_17)
target_include_directories(index_runtime_system_probe PRIVATE "${XR_SDK}/include")
target_link_libraries(index_runtime_system_probe PRIVATE "${XR_SDK}/lib/openxr_loader.lib")

add_executable(index_runtime_frame_probe index_runtime_frame_probe.cpp)
target_compile_features(index_runtime_frame_probe PRIVATE cxx_std_17)
target_include_directories(index_runtime_frame_probe PRIVATE "${XR_SDK}/include")
target_link_libraries(index_runtime_frame_probe PRIVATE "${XR_SDK}/lib/openxr_loader.lib" opengl32 gdi32 user32)

enable_testing()
add_test(NAME index_frame_probe_argument_guard COMMAND index_runtime_frame_probe)
set_tests_properties(index_frame_probe_argument_guard PROPERTIES WILL_FAIL TRUE)
add_test(NAME index_frame_probe_environment_guard COMMAND index_runtime_frame_probe --probe-frame)
set_tests_properties(index_frame_probe_environment_guard PROPERTIES WILL_FAIL TRUE)

find_package(Vulkan REQUIRED)
add_executable(vulkan_display_probe vulkan_display_probe.cpp)
target_compile_features(vulkan_display_probe PRIVATE cxx_std_17)
target_link_libraries(vulkan_display_probe PRIVATE Vulkan::Vulkan)
